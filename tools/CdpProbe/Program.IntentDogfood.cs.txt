var exe = args.ElementAtOrDefault(0) ?? @"D:\cdp-mcp\CdpMcp.exe";

await using var client = await McpClient.CreateAsync(new StdioClientTransport(new()
{
    Name = "intent-dogfood",
    Command = exe,
    Arguments = ["--config", config],
}));

async Task<string> Call(string name, Dictionary<string, object?> arguments)
{
    var result = await client.CallToolAsync(name, arguments);
    return string.Join("\n", result.Content.OfType<TextContentBlock>().Select(t => t.Text));
}

var tools = await client.ListToolsAsync();
var names = tools.Select(t => t.Name).ToHashSet(StringComparer.Ordinal);
if (!names.Contains("cdp_work") || !names.Contains("get_workspace_navigation_context"))
    throw new Exception("missing tools: " + string.Join(",", names.Order()));

Console.WriteLine(await Call("cdp_work", new() { ["op"] = "intent_upsert", ["title"] = "ts-cide-explore" }));
